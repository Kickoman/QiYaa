add_library(qiyaa_fdk_aac STATIC ${QIYAA_FDK_AAC_SOURCES})
target_include_directories(qiyaa_fdk_aac
    PUBLIC
        libAACdec/include
        libSYS/include
    PRIVATE
        libArithCoding/include
        libDRCdec/include
        libFDK/include
        libMpegTPDec/include
        libPCMutils/include
        libSACdec/include
        libSBRdec/include)
set_target_properties(qiyaa_fdk_aac PROPERTIES POSITION_INDEPENDENT_CODE ON)
# As upstream builds it, and quiet: third-party code is not held to QiYaa's warnings.
if(MSVC)
    target_compile_options(qiyaa_fdk_aac PRIVATE /EHsc /w)
else()
    target_compile_options(qiyaa_fdk_aac PRIVATE -fno-exceptions -fno-rtti -w)
endif()
if(UNIX)
    target_link_libraries(qiyaa_fdk_aac PRIVATE m)
endif()
